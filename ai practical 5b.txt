0/1 Knapsack using Dynamic Programming



w = [1, 2, 3]
v = [10, 15, 40]
cap = 5
n = len(w)

dp = [[0] * (cap+1) for i in range(n+1)]

for i in range(1, n+1):
    for c in range(1, cap+1):
        if w[i-1] <= c:
            dp[i][c] = max(v[i-1] + dp[i-1][c-w[i-1]],
                           dp[i-1][c])
        else:
            dp[i][c] = dp[i-1][c]

print("Max value:", dp[n][cap])
